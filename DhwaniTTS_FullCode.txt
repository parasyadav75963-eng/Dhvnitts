    private List<VoiceModel> currentVoicesForLang = new ArrayList<>();

    private ArrayAdapter<String> voiceAdapter;
    private List<String> voiceDisplayNames = new ArrayList<>();
    
    private ExecutorService executorService = Executors.newSingleThreadExecutor();
    private Handler mainHandler = new Handler(Looper.getMainLooper());

    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {

                updateVoiceSpinner(downloadedLanguages.get(position));
            }
            @Override public void onNothingSelected(AdapterView<?> parent) {}
        });

        loadSettings();

        seekbarSpeechRate.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) { tvSpeechRate.setText("Speech Rate: " + progress + "%"); }
            @Override public void onStartTrackingTouch(SeekBar seekBar) {}

    private void playRealPreview(String voiceKey, String text, float speed) {
        executorService.execute(() -> {
            try {
                File voicesDir = new File(requireContext().getFilesDir(), "voices");
                File modelFile = new File(voicesDir, voiceKey + ".onnx");
                File configFile = new File(voicesDir, voiceKey + ".onnx.json");

                if (modelFile.exists() && configFile.exists()) {
                    PiperEngine engine = new PiperEngine();
                    if (engine.initEngine(modelFile.getAbsolutePath(), configFile.getAbsolutePath())) {
                        byte[] audioData = engine.synthesizeText(text, speed);
                        if (audioData != null && audioData.length > 0) {
                            int minBuf = AudioTrack.getMinBufferSize(16000, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT);
                            AudioTrack audioTrack = new AudioTrack(AudioManager.STREAM_MUSIC, 16000, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT, Math.max(minBuf, audioData.length), AudioTrack.MODE_STREAM);
                            audioTrack.play();
                            audioTrack.write(audioData, 0, audioData.length);
                            audioTrack.stop();
                            audioTrack.release();
                        } else {
                            mainHandler.post(() -> Toast.makeText(getContext(), "Synthesis failed (Audio empty)", Toast.LENGTH_SHORT).show());
                        }
                        engine.destroyEngine();
                    } else {
                        mainHandler.post(() -> Toast.makeText(getContext(), "Engine Init Failed! File corrupt.", Toast.LENGTH_LONG).show());
                    }
                } else {
                    mainHandler.post(() -> Toast.makeText(getContext(), "Voice files missing! Please redownload.", Toast.LENGTH_SHORT).show());
                }
            } catch (Exception e) { e.printStackTrace(); }
        });
    }

    private void scanDownloadedVoices() {
        allDownloadedVoices.clear();
        downloadedLanguages.clear();
        File voicesDir = new File(requireContext().getFilesDir(), "voices");
        
        try {
            File cachedJson = new File(requireContext().getFilesDir(), "cached_voices.json");
            if (!cachedJson.exists()) {
                downloadedLanguages.add("Open Voice Library to sync");
                return;
            }

            BufferedReader br = new BufferedReader(new InputStreamReader(requireContext().openFileInput("cached_voices.json")));
            StringBuilder sb = new StringBuilder();
            String line;
            while((line = br.readLine()) != null) sb.append(line);
            br.close();

            JSONObject root = new JSONObject(sb.toString());
            File[] downloadedFiles = voicesDir.listFiles((dir, name) -> name.endsWith(".onnx"));
            
            if (downloadedFiles != null) {
                for (File file : downloadedFiles) {
                    String key = file.getName().replace(".onnx", "");
                    if (root.has(key)) {
                        JSONObject voiceObj = root.getJSONObject(key);
                        String langEnglish = voiceObj.getJSONObject("language").optString("name_english", "Unknown");
                        String voiceName = voiceObj.optString("name", "Unknown");
                        String quality = voiceObj.optString("quality", "");
                        
                        allDownloadedVoices.add(new VoiceModel(key, langEnglish, "", voiceName, quality, "", "", "", true));
                        if (!downloadedLanguages.contains(langEnglish)) {
                            downloadedLanguages.add(langEnglish);
                        }
                    }
                }
            }
        } catch (Exception e) { e.printStackTrace(); }

        if (downloadedLanguages.isEmpty()) { downloadedLanguages.add("No downloaded voices"); }
    }

    private void setupSpinners() {
        langAdapter = new ArrayAdapter<>(requireContext(), android.R.layout.simple_spinner_dropdown_item, downloadedLanguages);
        spinnerLanguage.setAdapter(langAdapter);
        String[] numReading = {"Default", "Digits", "Words"};
        spinnerNumber.setAdapter(new ArrayAdapter<>(requireContext(), android.R.layout.simple_spinner_dropdown_item, numReading));
        String[] punctuation = {"None", "Some", "Most", "All"};
        spinnerPunctuation.setAdapter(new ArrayAdapter<>(requireContext(), android.R.layout.simple_spinner_dropdown_item, punctuation));
    }

    private void updateVoiceSpinner(String selectedLang) {
        currentVoicesForLang.clear();
        voiceDisplayNames.clear();
        
        if (selectedLang.equals("No downloaded voices") || selectedLang.equals("Open Voice Library to sync")) {
            voiceDisplayNames.add("No voices available");
        } else {
            for (VoiceModel v : allDownloadedVoices) {
                if (v.getLanguageName().equals(selectedLang)) {
                    currentVoicesForLang.add(v);
                    voiceDisplayNames.add(v.getVoiceName() + " [" + v.getQuality() + "]");
                }
            }
        }
        voiceAdapter = new ArrayAdapter<>(requireContext(), android.R.layout.simple_spinner_dropdown_item, voiceDisplayNames);
        spinnerVoice.setAdapter(voiceAdapter);
    }

    private void saveSettings() {
        SharedPreferences.Editor editor = prefs.edit();
        editor.putInt("speech_rate", seekbarSpeechRate.getProgress());
        editor.putInt("pitch_rate", seekbarPitch.getProgress());
        editor.putBoolean("force_pitch", switchForcePitch.isChecked());
        editor.putBoolean("force_speech", switchForceSpeech.isChecked());
        
        int selectedVoicePos = spinnerVoice.getSelectedItemPosition();
        if (selectedVoicePos >= 0 && currentVoicesForLang.size() > selectedVoicePos) {
            VoiceModel selectedVoice = currentVoicesForLang.get(selectedVoicePos);
            // TTS service ke liye asali model key save kar rahe hain
            editor.putString("saved_voice_key", selectedVoice.getKey());
        }
        
        editor.putInt("num_pos", spinnerNumber.getSelectedItemPosition());
        editor.putInt("punc_pos", spinnerPunctuation.getSelectedItemPosition());
        editor.apply();
        Toast.makeText(getContext(), "Settings Saved as Default!", Toast.LENGTH_SHORT).show();
    }

    private void loadSettings() {
        seekbarSpeechRate.setProgress(prefs.getInt("speech_rate", 50));
        seekbarPitch.setProgress(prefs.getInt("pitch_rate", 50));
        switchForcePitch.setChecked(prefs.getBoolean("force_pitch", false));
        switchForceSpeech.setChecked(prefs.getBoolean("force_speech", false));
        spinnerNumber.setSelection(prefs.getInt("num_pos", 0));
        spinnerPunctuation.setSelection(prefs.getInt("punc_pos", 0));
    }

    private void resetSettings() {
        seekbarSpeechRate.setProgress(50);
        seekbarPitch.setProgress(50);
        switchForcePitch.setChecked(false);
        switchForceSpeech.setChecked(false);
        if (spinnerLanguage.getCount() > 0) spinnerLanguage.setSelection(0);
        spinnerNumber.setSelection(0);
        spinnerPunctuation.setSelection(0);
        saveSettings();
        Toast.makeText(getContext(), "Reset to Default", Toast.LENGTH_SHORT).show();
    }

    private void clearSettings() {
        prefs.edit().clear().apply();
        resetSettings();
        Toast.makeText(getContext(), "All Settings Cleared!", Toast.LENGTH_SHORT).show();
    }
}

public class LanguageAdapter extends RecyclerView.Adapter<LanguageAdapter.ViewHolder> {

    private List<String> languageList;
    private OnLanguageClickListener listener;

    public interface OnLanguageClickListener {
        void onLanguageClick(String lang);
    }

    public LanguageAdapter(List<String> languageList, OnLanguageClickListener listener) {
        this.languageList = languageList;
        this.listener = listener;
    }

    @NonNull
    @Override
    public ViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext()).inflate(android.R.layout.simple_list_item_1, parent, false);
        return new ViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull ViewHolder holder, int position) {
        String lang = languageList.get(position);
        holder.textView.setText(lang);
        
        holder.itemView.setOnClickListener(v -> {
            if (listener != null) {
                listener.onLanguageClick(lang);
            }
        });
    }

    @Override
    public int getItemCount() {
        return languageList != null ? languageList.size() : 0;
    }

    public static class ViewHolder extends RecyclerView.ViewHolder {
        TextView textView;

        public ViewHolder(@NonNull View itemView) {
            super(itemView);
            textView = itemView.findViewById(android.R.id.text1);
        }
    }
}

public class PiperEngine {

package com.itfb.dhwanitts;

import android.content.Context;
import android.content.SharedPreferences;
import android.media.AudioFormat;
import android.speech.tts.SynthesisCallback;
import android.speech.tts.SynthesisRequest;
import android.speech.tts.TextToSpeech;
import android.speech.tts.TextToSpeechService;
import android.text.TextUtils;
import android.util.Log;
import java.io.File;

public class TtsService extends TextToSpeechService {
    private static final String TAG = "DhwaniTTS_Service";
    private PiperEngine piperEngine;
    private SharedPreferences prefs;
    private String currentLoadedVoiceKey = ""; // मॉडल को बार-बार लोड होने से रोकने के लिए

    @Override
    public void onCreate() {
        super.onCreate();
        prefs = getSharedPreferences("DhwaniTTS_Prefs", Context.MODE_PRIVATE);
        piperEngine = new PiperEngine();
    }

    @Override
    protected int onIsLanguageAvailable(String lang, String country, String variant) {
        return TextToSpeech.LANG_AVAILABLE;
    }

    @Override
    protected String[] onGetLanguage() {
        return new String[]{"hin", "IND", ""};
    }

    @Override
    protected int onLoadLanguage(String lang, String country, String variant) {
        return TextToSpeech.LANG_AVAILABLE;
    }

    @Override
    protected void onStop() {
        Log.d(TAG, "TalkBack stopped speaking");
        // यहाँ इंजन को डिस्ट्रॉय न करें, वर्ना अगला वाक्य नहीं बोलेगा
    }

    @Override
    protected void onSynthesizeText(SynthesisRequest request, SynthesisCallback callback) {
        CharSequence textCharSequence = request.getCharSequenceText();
        if (TextUtils.isEmpty(textCharSequence)) {
            callback.done();
            return;
        }
        String text = textCharSequence.toString();

        String savedVoiceKey = prefs.getString("saved_voice_key", "");
        if (savedVoiceKey.isEmpty()) {
            callback.error();
            return;
        }

        File voicesDir = new File(getFilesDir(), "voices");
        File modelFile = new File(voicesDir, savedVoiceKey + ".onnx");
        File configFile = new File(voicesDir, savedVoiceKey + ".onnx.json");

        if (!modelFile.exists() || !configFile.exists()) {
            Log.e(TAG, "Model or Config missing!");
            callback.error();
            return;
        }

        // फिक्स: मॉडल केवल तभी लोड होगा जब वॉइस बदली गई हो (TalkBack स्पीड फिक्स)
        if (!savedVoiceKey.equals(currentLoadedVoiceKey)) {
            boolean isInit = piperEngine.initEngine(modelFile.getAbsolutePath(), configFile.getAbsolutePath());
            if (isInit) {
                currentLoadedVoiceKey = savedVoiceKey;
            } else {
                Log.e(TAG, "Piper Engine init failed!");
                callback.error();
                return;
            }
        }

        boolean forceSpeech = prefs.getBoolean("force_speech", false);
        float speed = forceSpeech ? (prefs.getInt("speech_rate", 50) / 50.0f) : (request.getSpeechRate() / 100.0f);

        // Piper C++ 16000Hz PCM जनरेट करता है
        callback.start(16000, AudioFormat.ENCODING_PCM_16BIT, 1);
        
        byte[] audioData = piperEngine.synthesizeText(text, speed);
        
        if (audioData != null && audioData.length > 0) {
            int maxBufferSize = callback.getMaxBufferSize();
            int offset = 0;
            while (offset < audioData.length) {
                int bytesToWrite = Math.min(maxBufferSize, audioData.length - offset);
                callback.audioAvailable(audioData, offset, bytesToWrite);
                offset += bytesToWrite;
            }
            callback.done();
        } else {
            Log.e(TAG, "Synthesis failed (Audio empty) - Espeak-ng data might be missing in C++!");
            callback.error();
        }
    }
    
    @Override
    public void onDestroy() {
        if (piperEngine != null) {
            piperEngine.destroyEngine();
            currentLoadedVoiceKey = "";
        }
        super.onDestroy();
    }
}

            holder.btnDownload.setText("Downloading...");

    private void playRealPreview(String modelPath, String configPath, String text, Context context) {
        executorService.execute(() -> {
            try {
                PiperEngine engine = new PiperEngine();
                if (engine.initEngine(modelPath, configPath)) {
                    byte[] audioData = engine.synthesizeText(text, 1.0f);
                    if (audioData != null && audioData.length > 0) {
                        int minBuf = AudioTrack.getMinBufferSize(16000, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT);
                        AudioTrack audioTrack = new AudioTrack(AudioManager.STREAM_MUSIC, 16000, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT, Math.max(minBuf, audioData.length), AudioTrack.MODE_STREAM);
                        audioTrack.play();
                        audioTrack.write(audioData, 0, audioData.length);
                        audioTrack.stop();
                        audioTrack.flush(); audioTrack.release();
                    } else {
                        mainHandler.post(() -> Toast.makeText(context, "Synthesis failed (Audio empty)", Toast.LENGTH_SHORT).show());
                    }
                    engine.destroyEngine();
                } else {
                    mainHandler.post(() -> Toast.makeText(context, "Engine Init Failed! File corrupt.", Toast.LENGTH_LONG).show());
                }
            } catch (Exception e) { e.printStackTrace(); }
        });
    }

    private void downloadVoiceFile(Context context, VoiceModel voice, int position) {
        executorService.execute(() -> {
            try {
                File voicesDir = new File(context.getFilesDir(), "voices");
                if (!voicesDir.exists()) voicesDir.mkdirs();
                File modelFile = new File(voicesDir, voice.getKey() + ".onnx");
                File configFile = new File(voicesDir, voice.getKey() + ".onnx.json");

                String modelUrl = BASE_URL + voice.getModelPath();
                String configUrl = BASE_URL + voice.getConfigPath();

                downloadSingleFile(modelUrl, modelFile);
                downloadSingleFile(configUrl, configFile);

                mainHandler.post(() -> {
                    voice.setDownloaded(true);
                    Toast.makeText(context, "Download Complete", Toast.LENGTH_SHORT).show();

                });

                    notifyItemChanged(position);
                });
            }
        });
    }

    private void downloadSingleFile(String fileUrl, File outputFile) throws Exception {
        URL url = new URL(fileUrl);
        HttpURLConnection connection;
        int redirects = 0;
        
        while (true) {
            connection = (HttpURLConnection) url.openConnection();
            connection.setInstanceFollowRedirects(false);
            connection.setRequestProperty("User-Agent", "Mozilla/5.0");
            
            int status = connection.getResponseCode();
            if (status == 301 || status == 302 || status == 303 || status == 307 || status == 308) {
                String newUrl = connection.getHeaderField("Location");
                url = new URL(url, newUrl);
                redirects++;
                if (redirects > 10) throw new Exception("Redirect Loop");
            } else if (status == HttpURLConnection.HTTP_OK) {
                break;
            } else {
                throw new Exception("HTTP Error: " + status);
            }
        }

        InputStream input = connection.getInputStream();
        FileOutputStream output = new FileOutputStream(outputFile);
        byte[] data = new byte[8192];
        int count;
        while ((count = input.read(data)) != -1) {
            output.write(data, 0, count);
        }
        output.flush();
        output.close();
        input.close();

        if (outputFile.length() < 1024) {
            outputFile.delete();
            throw new Exception("File Corrupt (Too Small)");
        }
    }

package com.itfb.dhwanitts;

import android.os.Handler;
import android.os.Looper;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class VoiceDownloader {
    public interface DownloadListener {
        void onProgress(int progress);
        void onSuccess();
        void onError(String error);
    }

    private static final ExecutorService executor = Executors.newSingleThreadExecutor();
    private static final Handler mainHandler = new Handler(Looper.getMainLooper());

    public static void downloadVoice(String fileUrl, String destinationPath, DownloadListener listener) {
        executor.execute(() -> {
            try {
                URL url = new URL(fileUrl);
                HttpURLConnection connection = (HttpURLConnection) url.openConnection();
                connection.connect();

                int responseCode = connection.getResponseCode();
                if (responseCode != HttpURLConnection.HTTP_OK) {
                    // रिस्पॉन्स कोड को पहले एक स्ट्रिंग में सेव कर रहे हैं ताकि Lambda में एरर न आए
                    final String errorMsg = "Server returned HTTP " + responseCode;
                    mainHandler.post(() -> listener.onError(errorMsg));
                    return;
                }

                int fileLength = connection.getContentLength();
                InputStream input = connection.getInputStream();
                OutputStream output = new FileOutputStream(destinationPath);

                byte[] data = new byte[4096];
                long total = 0;
                int count;
                while ((count = input.read(data)) != -1) {
                    total += count;
                    if (fileLength > 0) {
                        int progress = (int) (total * 100 / fileLength);
                        mainHandler.post(() -> listener.onProgress(progress));
                    }
                    output.write(data, 0, count);
                }
                output.flush();
                output.close();
                input.close();

                mainHandler.post(() -> listener.onSuccess());

            } catch (Exception e) {
                final String exMsg = e.getMessage() != null ? e.getMessage() : "Unknown network error";
                mainHandler.post(() -> listener.onError("Error: " + exMsg));
            }
        });
    }
}

    private void fetchOfficialVoices() {
        Toast.makeText(getContext(), "Syncing official Piper catalog...", Toast.LENGTH_SHORT).show();
        executorService.execute(() -> {
            try {
                URL url = new URL("https://huggingface.co/rhasspy/piper-voices/resolve/main/voices.json");
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("GET");
                conn.setConnectTimeout(15000);
                conn.setReadTimeout(15000);

                BufferedReader reader = new BufferedReader(new InputStreamReader(conn.getInputStream()));
                StringBuilder response = new StringBuilder();
                String line;
                while ((line = reader.readLine()) != null) response.append(line);
                reader.close();

                String jsonStr = response.toString();
                
                FileOutputStream fos = requireContext().openFileOutput("cached_voices.json", Context.MODE_PRIVATE);
                fos.write(jsonStr.getBytes());
                fos.close();

                parseVoicesJson(jsonStr);
            } catch (Exception e) {
                mainHandler.post(() -> Toast.makeText(getContext(), "Network Error. Loading cache.", Toast.LENGTH_LONG).show());
                loadFromCache();
            }
        });
    }

    private void loadFromCache() {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(requireContext().openFileInput("cached_voices.json")));
            StringBuilder sb = new StringBuilder();
            String line;
            while((line = br.readLine()) != null) sb.append(line);
            br.close();
            parseVoicesJson(sb.toString());
        } catch (Exception e) {
            mainHandler.post(() -> Toast.makeText(getContext(), "No cached data. Please connect to internet.", Toast.LENGTH_LONG).show());
        }
    }

    private void parseVoicesJson(String jsonStr) {
        try {
            JSONObject root = new JSONObject(jsonStr);
            Iterator<String> keys = root.keys();
            List<VoiceModel> fetchedVoices = new ArrayList<>();
            File voicesDir = new File(requireContext().getFilesDir(), "voices");

            while (keys.hasNext()) {
                String key = keys.next();
                JSONObject voiceObj = root.getJSONObject(key);
                
                JSONObject langObj = voiceObj.getJSONObject("language");
                String langEnglish = langObj.optString("name_english", "Unknown");
                String country = langObj.optString("country_english", "");
                String quality = voiceObj.optString("quality", "medium");
                String voiceName = voiceObj.optString("name", "Unknown");

                JSONObject filesObj = voiceObj.getJSONObject("files");
                String modelPath = "";
                String configPath = "";
                long totalSizeBytes = 0;

                Iterator<String> fileKeys = filesObj.keys();
                while (fileKeys.hasNext()) {
                    String filePath = fileKeys.next();
                    if (filePath.endsWith(".onnx")) modelPath = filePath;
                    else if (filePath.endsWith(".json")) configPath = filePath;
                    
                    JSONObject fileMeta = filesObj.getJSONObject(filePath);
                    totalSizeBytes += fileMeta.optLong("size_bytes", 0);
                }

                if (modelPath.isEmpty() || configPath.isEmpty()) continue;

                String sizeMB = String.format("%.1f MB", totalSizeBytes / (1024.0 * 1024.0));
                
                File checkModel = new File(voicesDir, key + ".onnx");
                File checkConfig = new File(voicesDir, key + ".onnx.json");
                boolean isDownloaded = checkModel.exists() && checkConfig.exists() && checkModel.length() > 50000;

                fetchedVoices.add(new VoiceModel(key, langEnglish, country, voiceName, quality, modelPath, configPath, sizeMB, isDownloaded));
            }

            mainHandler.post(() -> {
                voiceList.clear();
                voiceList.addAll(fetchedVoices);
                adapter.notifyDataSetChanged();
                Toast.makeText(getContext(), "Voices Loaded!", Toast.LENGTH_SHORT).show();
            });
        } catch (Exception e) { e.printStackTrace(); }
    }
}

public class VoiceModel {
    private String key;
    private String languageName;
    private String country;
    private String voiceName;
    private String quality;
    private String modelPath;
    private String configPath;
    private String sizeMB;
    private boolean isDownloaded;

    public VoiceModel(String key, String languageName, String country, String voiceName, String quality, String modelPath, String configPath, String sizeMB, boolean isDownloaded) {
        this.key = key;
        this.languageName = languageName;
        this.country = country;
        this.voiceName = voiceName;
        this.quality = quality;
        this.modelPath = modelPath;
        this.configPath = configPath;
        this.sizeMB = sizeMB;
        this.isDownloaded = isDownloaded;
    }

    public String getKey() { return key; }
    public String getLanguageName() { return languageName; }
    public String getCountry() { return country; }
    public String getVoiceName() { return voiceName; }
    public String getQuality() { return quality; }
    public String getModelPath() { return modelPath; }
    public String getConfigPath() { return configPath; }
    public String getSizeMB() { return sizeMB; }
    public boolean isDownloaded() { return isDownloaded; }
    public void setDownloaded(boolean downloaded) { isDownloaded = downloaded; }
}

package com.itfb.dhwanitts;

import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.widget.Button;
import androidx.appcompat.app.AppCompatActivity;

public class WelcomeActivity extends AppCompatActivity {
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        // चेक करें कि क्या यूजर पहले ही ऐप खोल चुका है
        SharedPreferences prefs = getSharedPreferences("DhwaniTTS_Prefs", MODE_PRIVATE);
        boolean isFirstLaunch = prefs.getBoolean("isFirstLaunch", true);

        if (!isFirstLaunch) {
            // अगर पहली बार नहीं है, तो सीधे MainActivity पर जाएं
            startActivity(new Intent(this, MainActivity.class));
            finish();
            return;
        }

        setContentView(R.layout.activity_welcome);

        Button btnOk = findViewById(R.id.btn_ok_welcome);
        btnOk.setOnClickListener(v -> {
            // पहली बार का स्टेटस सेव करें
            prefs.edit().putBoolean("isFirstLaunch", false).apply();
            
            // MainActivity पर जाएं
            startActivity(new Intent(WelcomeActivity.this, MainActivity.class));
            finish();
        });
    }
}
